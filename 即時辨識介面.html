const videoElement = document.getElementById('input_video'); const canvasElement = document.getElementById('output_canvas'); const canvasCtx = canvasElement.getContext('2d'); const resultDiv = document.getElementById('result'); const startBtn = document.getElementById('startBtn'); const stopBtn = document.getElementById('stopBtn'); let stream = null; let animationFrameId = null; let holistic = null; let ws = null; let isProcessing = false; let isHolisticReady = false; // 💡 確保模型完全初始化完成的標記 let lastProcessingTime = 0; const TARGET_FPS = 20; const FRAME_INTERVAL = 1000 / TARGET_FPS; // 1. POSE 關鍵點 (6個點 * 4值 = 24維) const POSE_INDICES = [11, 12, 13, 14, 15, 16]; // 2. 對齊 config.py 的 FACE_INDEX 字典定義 const FACE_INDEX_CONFIG = { "right_eyebrow": [46, 52, 55, 107, 105, 70], "left_eyebrow": [276, 282, 285, 336, 334, 300], "right_eye": [33, 145, 133, 158, 159], "left_eye": [362, 374, 263, 386, 385], "nose": [6, 129, 98, 2, 327, 358], "mouth": [61, 91, 17, 321, 291, 267, 0, 37], "forehead": [9, 284, 297, 10, 67, 54], "chin": [152, 400, 431, 406, 18, 182, 211, 176] }; // 將 config.py 的 FACE_INDEX 展開成陣列清單 const ALL_CONFIG_FACE_INDICES = Object.values(FACE_INDEX_CONFIG).flat(); // 取前 16 個點填補至 198 維特徵中 (16點 * 3 = 48維) const FEATURE_FACE_INDICES = ALL_CONFIG_FACE_INDICES.slice(0, 16); async function initMediaPipe() { if (holistic) return; holistic = new Holistic({ locateFile: (file) => `https://cdn.jsdelivr.net/npm/@mediapipe/holistic@0.5.1675471629/${file}` }); holistic.setOptions({ modelComplexity: 0, smoothLandmarks: true, minDetectionConfidence: 0.3, minTrackingConfidence: 0.3, refineLandmarks: true }); holistic.onResults((results) => { try { canvasCtx.save(); canvasCtx.clearRect(0, 0, canvasElement.width, canvasElement.height); canvasCtx.drawImage(results.image, 0, 0, canvasElement.width, canvasElement.height); const width = canvasElement.width; const height = canvasElement.height; // 1. 🎯 肩膀與手臂特徵點 (6點 - 亮橘色) if (results.poseLandmarks) { canvasCtx.fillStyle = "#FF9900"; for (let i = 0; i < POSE_INDICES.length; i++) { const lm = results.poseLandmarks[POSE_INDICES[i]]; if (lm) { canvasCtx.beginPath(); canvasCtx.arc(lm.x * width, lm.y * height, 4, 0, 2 * Math.PI); canvasCtx.fill(); } } } // 2. 🎯 左手掌特徵點 (21點 - 青色) if (results.leftHandLandmarks && results.leftHandLandmarks.length > 0) { canvasCtx.fillStyle = "#00FFCC"; for (let i = 0; i < results.leftHandLandmarks.length; i++) { const lm = results.leftHandLandmarks[i]; if (lm) { canvasCtx.beginPath(); canvasCtx.arc(lm.x * width, lm.y * height, 3.5, 0, 2 * Math.PI); canvasCtx.fill(); } } } // 3. 🎯 右手掌特徵點 (21點 - 黃綠色) if (results.rightHandLandmarks && results.rightHandLandmarks.length > 0) { canvasCtx.fillStyle = "#38EF7D"; for (let i = 0; i < results.rightHandLandmarks.length; i++) { const lm = results.rightHandLandmarks[i]; if (lm) { canvasCtx.beginPath(); canvasCtx.arc(lm.x * width, lm.y * height, 3.5, 0, 2 * Math.PI); canvasCtx.fill(); } } } canvasCtx.restore(); // -------------------------------------------------------- // 198 維特徵打包 (對齊 FastAPI 後端模型) // -------------------------------------------------------- let landmarksData = new Array(198).fill(0); let idx = 0; // POSE (24維) if (results.poseLandmarks) { for (let i = 0; i < POSE_INDICES.length; i++) { const lm = results.poseLandmarks[POSE_INDICES[i]]; if (lm) { landmarksData[idx++] = lm.x; landmarksData[idx++] = lm.y; landmarksData[idx++] = lm.z; landmarksData[idx++] = lm.visibility || 0; } else { idx += 4; } } } else { idx += 24; } // LEFT HAND (63維) if (results.leftHandLandmarks) { for (let i = 0; i < results.leftHandLandmarks.length; i++) { const lm = results.leftHandLandmarks[i]; landmarksData[idx++] = lm.x; landmarksData[idx++] = lm.y; landmarksData[idx++] = lm.z; } } else { idx += 63; } // RIGHT HAND (63維) if (results.rightHandLandmarks) { for (let i = 0; i < results.rightHandLandmarks.length; i++) { const lm = results.rightHandLandmarks[i]; landmarksData[idx++] = lm.x; landmarksData[idx++] = lm.y; landmarksData[idx++] = lm.z; } } else { idx += 63; } // FACE (48維 - 使用 FEATURE_FACE_INDICES 打包) if (results.faceLandmarks) { for (let i = 0; i < FEATURE_FACE_INDICES.length; i++) { const lm = results.faceLandmarks[FEATURE_FACE_INDICES[i]]; if (lm) { landmarksData[idx++] = lm.x; landmarksData[idx++] = lm.y; landmarksData[idx++] = lm.z; } else { idx += 3; } } } else { idx += 48; } while (idx < 198) landmarksData[idx++] = 0; landmarksData = landmarksData.slice(0, 198); // 💡 持續發送資料至 WebSocket if (ws && ws.readyState === WebSocket.OPEN) { ws.send(JSON.stringify({ landmarks: landmarksData })); } } catch (err) { console.error("Canvas 繪圖異常:", err); } finally { isProcessing = false; } }); // 💡 務必呼叫 initialize() 以防 loadGraph 崩潰 await holistic.initialize(); isHolisticReady = true; } async function processFrame() { if (!stream || !isHolisticReady) return; const now = Date.now(); if (!isProcessing && (now - lastProcessingTime >= FRAME_INTERVAL)) { if (videoElement.readyState >= 2 && videoElement.videoWidth > 0 && videoElement.videoHeight > 0) { isProcessing = true; lastProcessingTime = now; try { await holistic.send({ image: videoElement }); } catch (err) { console.error("MediaPipe 傳送畫面異常:", err); isProcessing = false; } } } animationFrameId = requestAnimationFrame(processFrame); } async function startCamera() { try { resultDiv.innerText = "正在連線後端..."; resultDiv.style.color = "#00f2fe"; ws = new WebSocket("wss://sign-language-backend-qr50.onrender.com/ws/predict"); ws.onopen = () => { resultDiv.innerText = "已連線至後端，初始化模型中..."; resultDiv.style.color = "#00ffcc"; }; ws.onmessage = (event) => { try { const data = JSON.parse(event.data); resultDiv.innerText = `${data.prediction} ${data.confidence ? '(' + data.confidence + '%)' : ''}`; resultDiv.style.color = "#00f2fe"; } catch (e) { console.error("解析回應錯誤:", e); } }; ws.onerror = (error) => { console.error("WebSocket 錯誤:", error); resultDiv.innerText = "後端連線失敗！請確認 Render 是否已甦醒"; resultDiv.style.color = "#ff3333"; }; ws.onclose = () => { console.log("WebSocket 連線關閉"); }; // 💡 優先初始化 MediaPipe 模型 await initMediaPipe(); stream = await navigator.mediaDevices.getUserMedia({ video: { width: { ideal: 640 }, height: { ideal: 480 } }, audio: false }); videoElement.srcObject = stream; await videoElement.play(); startBtn.disabled = true; stopBtn.disabled = false; processFrame(); } catch (error) { console.error("啟動失敗:", error); resultDiv.innerText = "相機開啟或模型初始化失敗"; resultDiv.style.color = "#ff3333"; } } function stopCamera() { if (animationFrameId) { cancelAnimationFrame(animationFrameId); animationFrameId = null; } if (stream) { stream.getTracks().forEach(track => track.stop()); videoElement.srcObject = null; stream = null; } if (ws) { ws.close(); ws = null; } isProcessing = false; isHolisticReady = false; canvasCtx.clearRect(0, 0, canvasElement.width, canvasElement.height); resultDiv.innerText = "鏡頭未開啟"; resultDiv.style.color = "#00f2fe"; startBtn.disabled = false; stopBtn.disabled = true; }
